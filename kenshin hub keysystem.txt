-- دالة التحقق الرسمية عبر مكتبة Panda V4
local function validateKey(userKey)
    if not PUSL or type(PUSL.validate) ~= "function" then
        return false
    end
    
    local success, result = pcall(function()
        return PUSL.validate(userKey)
    end)
    
    if success and result then
        return result.success == true
    end
    return false
end

local function parseValue(val)
    if typeof(val) == "number" then
        return val
    elseif typeof(val) == "string" then
        local str = val:gsub(",", ""):gsub(" ", ""):upper()
        local num = tonumber(str:match("[%d%.]+")) or 0
        if str:find("DC") then return num * 1e33
        elseif str:find("NO") then return num * 1e30
        elseif str:find("OC") then return num * 1e27
        elseif str:find("SP") then return num * 1e24
        elseif str:find("SX") then return num * 1e21
        elseif str:find("QI") then return num * 1e18
        elseif str:find("QA") then return num * 1e15
        elseif str:find("T") then return num * 1e12
        elseif str:find("B") then return num * 1e9
        elseif str:find("M") then return num * 1e6
        elseif str:find("K") then return num * 1e3
        else return tonumber(str) or 0
        end
    end
    return 0
end

local function getPlayerDurability(player)
    local durabilityStat = nil
    if player:FindFirstChild("Durability") then
        durabilityStat = player.Durability
    elseif player:FindFirstChild("leaderstats") and player.leaderstats:FindFirstChild("Durability") then
        durabilityStat = player.leaderstats.Durability
    else
        for _, child in ipairs(player:GetChildren()) do
            if child:IsA("Folder") and child:FindFirstChild("Durability") then
                durabilityStat = child.Durability
                break
            end
        end
    end
    if durabilityStat then
        return parseValue(durabilityStat.Value)
    end
    return 0
end
